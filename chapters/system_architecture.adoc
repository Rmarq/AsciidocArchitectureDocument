==== Example of plantuml diagram integration.

[plantuml, subs="attributes"]
// ", subs="attributes"" allows asciidoc to replace variables inside the diagram like in this example, {projname} will be replaced when previewed from main.adoc
----
' Define actors
actor User

' Define system components
rectangle "{projname} Server" as Server
rectangle API
package Module1 {
    rectangle Server1
    rectangle Back1
}
package Module2 {
    rectangle Server2
    rectangle Back2
}

' Relationships
User --> Server
Server --> API
API -> Server1
API -> Server2
Server1 -> Back1
Server2 -> Back2


' Layout constraints (invisible)
Server1 -[hidden]d-> Server2
Server -[hidden]r-> Server1
Server -[hidden]r-> Server2
----
